// 規格化 Yahoo 代碼至 Google Finance 專用代碼
function mapSymbolToGoogleFinance(symbol) {
  const s = symbol.toUpperCase().trim();
  
  // 1. 匯率 (如 TWD=X -> CURRENCY:USDTWD)
  if (s.endsWith('=X')) {
    const currency = s.replace('=X', '');
    if (currency === 'TWD') return 'CURRENCY:USDTWD';
    if (currency === 'JPY') return 'CURRENCY:USDJPY';
    if (currency.length === 3) return 'CURRENCY:USD' + currency;
    return 'CURRENCY:' + currency;
  }
  if (s === 'EURUSD=X' || s === 'EURUSD') {
    return 'CURRENCY:EURUSD';
  }
  
  // 2. 指數與總經
  if (s === '^GSPC') return 'INDEXSP:.INX';
  if (s === '^IXIC') return 'INDEXNASDAQ:.IXIC';
  if (s === '^SOX') return 'INDEXNASDAQ:SOX';
  if (s === '^TNX') return 'INDEXCBOE:TNX';
  if (s === '^TWII') return 'TPE:TAIEX';
  
  // 3. 台灣個股 (如 2330.TW -> TPE:2330)
  if (s.endsWith('.TW')) {
    return 'TPE:' + s.replace('.TW', '');
  }
  
  // 4. 純數字代碼 (如台股代碼 2330 -> TPE:2330)
  if (/^\d+$/.test(s)) {
    return 'TPE:' + s;
  }
  
  // 5. 美股個股直接回傳
  return s;
}

// 透過多重爬蟲 (Histock + CnYes) 獲取台股報價 (確保一定抓得到)
function fetchTaiwanQuotesHTML(symbols, includeHistory) {
  const results = {};
  if (!symbols || symbols.length === 0) return results;

  for (let i = 0; i < symbols.length; i++) {
    const s = symbols[i];
    const match = s.match(/^([A-Z0-9]+)(\.TW|\.TWO)?$/);
    const rawSym = match ? match[1] : s.replace('.TWO', '').replace('.TW', '');
    
    let price = 0, change = 0, pct = 0, name = s;
    let success = false, errMsg = "";
    
    // 1. 優先嘗試 CnYes 鉅亨網 (因為有明確的漲跌正負號)
    try {
      const url2 = 'https://www.cnyes.com/twstock/' + rawSym;
      const res2 = UrlFetchApp.fetch(url2, { muteHttpExceptions: true, headers: { 'User-Agent': 'Mozilla/5.0' } });
      if (res2.getResponseCode() === 200) {
        const html2 = res2.getContentText();
        const descMatch2 = html2.match(/即時行情\s+([0-9.,]+)\s+([+-]?[0-9.]+)%/);
        if (descMatch2) {
          price = parseFloat(descMatch2[1].replace(/,/g, ''));
          pct = parseFloat(descMatch2[2]);
          const prevClose = price / (1 + pct / 100);
          change = price - prevClose;
          const nameMatch2 = html2.match(/content="([^"(]+)\([0-9]+\)\s*即時行情/);
          if (nameMatch2) name = nameMatch2[1].trim();
          success = true;
        } else {
          errMsg += "CnYes parse failed. ";
        }
      } else {
        errMsg += "CnYes HTTP " + res2.getResponseCode() + ". ";
      }
    } catch(e) { errMsg += "CnYes err: " + e.message + ". "; }

    // 2. 若 CnYes 失敗，備用 Histock (並嘗試抓取漲跌符號)
    if (!success) {
      try {
        const url = 'https://histock.tw/stock/' + rawSym;
        const res = UrlFetchApp.fetch(url, { muteHttpExceptions: true, headers: { 'User-Agent': 'Mozilla/5.0' } });
        if (res.getResponseCode() === 200) {
          const html = res.getContentText();
          const descMatch = html.match(/<meta name="description" content="[^0-9]*([0-9.,]+)[^0-9]*([+-]?[0-9.,]+[漲跌]*)[^0-9]*([0-9.,]+%)/);
          if (descMatch) {
            price = parseFloat(descMatch[1].replace(/,/g, ''));
            
            // 嘗試從 HTML 內尋找精確的漲跌百分比 (包含正負號)
            const pctMatch = html.match(/id="Price1_lbTPercent"[^>]*><span[^>]*>([+-]?[0-9.]+)%<\/span>/);
            if (pctMatch) {
              pct = parseFloat(pctMatch[1]);
              const prevClose = price / (1 + pct / 100);
              change = price - prevClose;
            } else {
              let rawChange = descMatch[2].replace(/,/g, '');
              let chgVal = parseFloat(rawChange.replace(/[^0-9.]/g, '')) || 0;
              change = chgVal;
              if (html.includes('id="Price1_lbTChange"><span class="clr-gr"')) {
                change = -chgVal;
              }
              pct = price > 0 ? (change / (price - change)) * 100 : 0;
            }
            
            const nameMatch = html.match(/<title>([^ ]+)\s*\([0-9]+\)/);
            if (nameMatch) name = nameMatch[1].trim();
            success = true;
          } else {
            errMsg += "Histock parse failed.";
          }
        } else {
          errMsg += "Histock HTTP " + res.getResponseCode() + ".";
        }
      } catch(e) { errMsg += "Histock err: " + e.message + "."; }
    }
    
    if (success) {
      let historyArray = [];
      if (includeHistory) {
        try {
          const today = new Date();
          const past30 = new Date(today.getTime() - 30 * 24 * 60 * 60 * 1000);
          const yyyy = past30.getFullYear();
          const mm = String(past30.getMonth() + 1).padStart(2, '0');
          const dd = String(past30.getDate()).padStart(2, '0');
          const startDate = `${yyyy}-${mm}-${dd}`;
          
          const historyUrl = `https://api.finmindtrade.com/api/v4/data?dataset=TaiwanStockPrice&data_id=${rawSym}&start_date=${startDate}`;
          const historyRes = UrlFetchApp.fetch(historyUrl, { muteHttpExceptions: true });
          if (historyRes.getResponseCode() === 200) {
            const histData = JSON.parse(historyRes.getContentText());
            if (histData && histData.data && histData.data.length > 0) {
              historyArray = histData.data.map(d => d.close);
            }
          }
        } catch(e) { console.log("FinMind err: " + e.message); }
      }
      
      results[s] = {
        status: 'success', price: price, change: change, changePercent: pct,
        longName: name, history: historyArray, volumeSpike: false
      };
    } else {
      results[s] = { status: 'error', message: errMsg };
    }
  }
  return results;
}

// 處理前端發送的 GET 請求
function doGet(e) {
  setup();
  
  const action = e.parameter.action;
  
  if (action === 'getChartHistory') {
    const symbol = (e.parameter.symbol || '').toUpperCase().trim();
    if (!symbol) return ContentService.createTextOutput(JSON.stringify({error: 'No symbol'})).setMimeType(ContentService.MimeType.JSON);
    
    let dates = [];
    let prices = [];
    
    if (symbol.endsWith('.TW') || symbol.endsWith('.TWO')) {
      // Taiwan Stocks: We no longer fetch FinMind in backend to avoid IP rate limits.
      // The frontend will handle this directly.
      return ContentService.createTextOutput(JSON.stringify({error: 'Taiwan stock history is handled by frontend'}))
        .setMimeType(ContentService.MimeType.JSON);
    } else {
      // Others: try to get from cache (already fetched by Google Finance)
      const cache = CacheService.getScriptCache();
      const hStr = cache.get('quote_hist_' + symbol);
      if (hStr) {
        prices = JSON.parse(hStr); // it's either an array or comma-separated string, actually wait
        if (!hStr.startsWith('[')) {
          prices = hStr.split(',').map(Number);
        } else {
          prices = JSON.parse(hStr);
        }
        // Generate mock dates for the past N days (weekdays)
        let d = new Date();
        for (let i = 0; i < prices.length; i++) {
          dates.unshift(d.toISOString().split('T')[0]);
          d.setDate(d.getDate() - 1);
          if (d.getDay() === 0) d.setDate(d.getDate() - 2); // skip weekend roughly
          else if (d.getDay() === 6) d.setDate(d.getDate() - 1);
        }
      }
    }
    
    return ContentService.createTextOutput(JSON.stringify({ symbol: symbol, dates: dates, prices: prices }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  
  // 1. 處理取得即時報價的 API 請求 (使用 Google Finance 引擎)
  if (action === 'getQuotes') {
    const symbolsStr = e.parameter.symbols || '';
    const symbols = symbolsStr.split(',').map(s => s.trim()).filter(s => s.length > 0);
    const includeHistory = e.parameter.includeHistory !== 'false';
    
    const results = {};
    if (symbols.length === 0) {
      return ContentService.createTextOutput(JSON.stringify({ quotes: {} }))
                           .setMimeType(ContentService.MimeType.JSON);
    }
    
    const cache = CacheService.getScriptCache();
    const missingSymbols = [];
    
    // 優先從 CacheService 讀取緩存的資料 (價格 10 秒，歷史 2 小時)
    symbols.forEach(s => {
      const priceCached = cache.get('quote_price_' + s);
      const histCached = includeHistory ? cache.get('quote_hist_' + s) : null;
      
      if (priceCached && (!includeHistory || histCached)) {
        try {
          const pObj = JSON.parse(priceCached);
          const hObj = histCached ? JSON.parse(histCached) : [];
          results[s] = {
            status: 'success',
            price: pObj.price,
            change: pObj.change,
            changePercent: pObj.changePercent,
            longName: pObj.longName,
            volumeSpike: pObj.volumeSpike,
            history: hObj
          };
        } catch (err) {
          missingSymbols.push(s);
        }
      } else {
        missingSymbols.push(s);
      }
    });
    
    // 分流：台灣股票 (包含 .TW 與 .TWO) 全部交由網頁爬蟲處理，其餘交由 Google Finance
    const yfSymbols = missingSymbols.filter(s => s.endsWith('.TW') || s.endsWith('.TWO'));
    const gfSymbols = missingSymbols.filter(s => !s.endsWith('.TW') && !s.endsWith('.TWO'));
    const toCache = {};

    if (yfSymbols.length > 0) {
      const yfResults = fetchTaiwanQuotesHTML(yfSymbols, includeHistory);
      for (let i = 0; i < yfSymbols.length; i++) {
        const sym = yfSymbols[i];
        const data = yfResults[sym];
        if (data && data.status === 'success') {
          results[sym] = data;
          toCache['quote_price_' + sym] = JSON.stringify({
            price: data.price, change: data.change, changePercent: data.changePercent,
            longName: data.longName, volumeSpike: data.volumeSpike
          });
          if (includeHistory && data.history && data.history.length > 0) {
            cache.put('quote_hist_' + sym, JSON.stringify(data.history), 7200);
          }
        } else {
          results[sym] = data || { status: 'error', message: 'Yahoo API failed' };
        }
      }
    }

    if (gfSymbols.length > 0) {
      const lock = LockService.getScriptLock();
      try {
        // 等待最長 15 秒以取得腳本鎖，避免多個網頁籤/使用者併發導致覆寫 cacheSheet
        lock.waitLock(15000);
        
        const ss = SpreadsheetApp.getActiveSpreadsheet();
        let cacheSheet = ss.getSheetByName('__Quotes_Cache__');
        if (!cacheSheet) {
          cacheSheet = ss.insertSheet('__Quotes_Cache__');
        }
        
        // 重設與配置快取工作表
        cacheSheet.clear();
        cacheSheet.getRange(1, 1, 1, 8).setValues([['Symbol', 'Price', 'Change', 'ChangePct', 'Name', 'History', 'Volume', 'VolumeAvg20']]);
        
        // 寫入規格化代碼至 Column A
        const mappedSymbols = gfSymbols.map(s => [mapSymbolToGoogleFinance(s)]);
        cacheSheet.getRange(2, 1, gfSymbols.length, 1).setValues(mappedSymbols);
        
        // 寫入 Google Finance 計算公式至 Column B 至 H
        const formulas = [];
        for (let i = 0; i < gfSymbols.length; i++) {
          const rowIdx = i + 2;
          formulas.push([
            `=GOOGLEFINANCE(A${rowIdx})`,
            `=GOOGLEFINANCE(A${rowIdx}, "change")`,
            `=GOOGLEFINANCE(A${rowIdx}, "changepct")`,
            `=GOOGLEFINANCE(A${rowIdx}, "name")`,
            includeHistory ? `=JOIN(";", INDEX(GOOGLEFINANCE(A${rowIdx}, "close", TODAY()-730, TODAY()), 0, 2))` : "",
            `=GOOGLEFINANCE(A${rowIdx}, "volume")`,
            includeHistory ? `=AVERAGE(INDEX(GOOGLEFINANCE(A${rowIdx}, "volume", TODAY()-30, TODAY()), 0, 2))` : ""
          ]);
        }
        cacheSheet.getRange(2, 2, gfSymbols.length, 7).setFormulas(formulas);
        
        // 輪詢等待公式在 Google 伺服器端計算完畢 (最多嘗試 15 次，每次 500ms，以解決個股加載慢的問題)
        let values = [];
        const dataRange = cacheSheet.getRange(2, 1, gfSymbols.length, 8);
        for (let attempt = 0; attempt < 15; attempt++) {
          SpreadsheetApp.flush();
          values = dataRange.getValues();
          
          let hasLoading = false;
          for (let i = 0; i < values.length; i++) {
            const val = String(values[i][1]); // 價格
            const histVal = String(values[i][5]); // 歷史走勢
            
            const isHistLoading = includeHistory && (histVal === "" || histVal.toUpperCase().includes("LOADING"));
            
            // 空白或包含 loading 代表仍在計算中，需要等待；若為 #N/A 代表計算完成且不支援該欄位
            if (val === "" || val.toUpperCase().includes("LOADING") || isHistLoading) {
              hasLoading = true;
              break;
            }
          }
          if (!hasLoading) {
            break;
          }
          Utilities.sleep(500);
        }
        
        // 解析並建立快取
        for (let i = 0; i < gfSymbols.length; i++) {
          const origSym = gfSymbols[i];
          const rowValues = values[i];
          
          let rawPriceStr = String(rowValues[1]).trim();
          // 支援歐式小數點 (123,45) 格式與千分位逗號清理
          if (rawPriceStr.includes(',') && !rawPriceStr.includes('.')) {
            rawPriceStr = rawPriceStr.replace(/,/g, '.');
          } else {
            rawPriceStr = rawPriceStr.replace(/,/g, '');
          }
          rawPriceStr = rawPriceStr.replace(/[^0-9.-]/g, '');
          const priceVal = parseFloat(rawPriceStr);
          
          if (isNaN(priceVal) || priceVal <= 0) {
            results[origSym] = { status: 'error', message: 'Price loading failed' };
            continue;
          }
          
          const name = rowValues[4] || origSym;
          
          // 解析歷史資料
          let historyArr = [];
          if (includeHistory) {
            const historyStr = String(rowValues[5] || '');
            historyArr = historyStr.split(';').map(s => {
              let cleaned = s.trim().replace(/,/g, '.');
              return parseFloat(cleaned);
            }).filter(p => !isNaN(p) && p > 0);
          } else {
            const histCached = cache.get('quote_hist_' + origSym);
            if (histCached) {
              try { historyArr = JSON.parse(histCached); } catch (e) {}
            }
          }
          
          let rawChange = String(rowValues[2] || '').trim().replace(/,/g, '.').replace(/[^0-9.-]/g, '');
          let change = parseFloat(rawChange);
          
          let rawChangePercent = String(rowValues[3] || '').trim().replace(/,/g, '.').replace(/[^0-9.-]/g, '');
          let changePercent = parseFloat(rawChangePercent);
          
          // 外匯或部分商品不原生支援 change，則以 30 天歷史收盤價的前一天價格做計算
          if ((isNaN(change) || change === 0) && historyArr.length >= 2) {
            const prevPrice = historyArr[historyArr.length - 2];
            change = priceVal - prevPrice;
            changePercent = (change / prevPrice) * 100;
          }
          
          // 爆量檢測 (本日成交量 > 2.5倍 的 20日均量)
          let rawVolume = String(rowValues[6] || '').trim().replace(/,/g, '').replace(/[^0-9.-]/g, '');
          const volume = parseFloat(rawVolume) || 0;
          
          let rawVolumeAvg = String(rowValues[7] || '').trim().replace(/,/g, '').replace(/[^0-9.-]/g, '');
          const volumeAvg = parseFloat(rawVolumeAvg) || 0;
          const volumeSpike = volumeAvg > 0 && volume > 2.5 * volumeAvg;
          
          const quoteObj = {
            status: 'success',
            price: priceVal,
            change: isNaN(change) ? 0 : change,
            changePercent: isNaN(changePercent) ? 0 : changePercent,
            longName: name,
            history: historyArr,
            volumeSpike: volumeSpike
          };
          
          results[origSym] = quoteObj;
          
          // 快取價格與基本欄位 (10秒效期)
          toCache['quote_price_' + origSym] = JSON.stringify({
            price: priceVal,
            change: isNaN(change) ? 0 : change,
            changePercent: isNaN(changePercent) ? 0 : changePercent,
            longName: name,
            volumeSpike: volumeSpike
          });
          
          // 快取歷史收盤價 (2小時效期)
          if (includeHistory && historyArr.length > 0) {
            cache.put('quote_hist_' + origSym, JSON.stringify(historyArr), 7200);
          }
        }
        
      } catch (err) {
        // 若發生錯誤，為防阻斷，仍對未成功的標的填入錯誤狀態
        gfSymbols.forEach(s => {
          if (!results[s]) {
            results[s] = { status: 'error', message: 'Exception: ' + err.toString() };
          }
        });
      } finally {
        lock.releaseLock();
      }
    }

    if (Object.keys(toCache).length > 0) {
      cache.putAll(toCache, 10);
    }
    
    return ContentService.createTextOutput(JSON.stringify({ quotes: results }))
                         .setMimeType(ContentService.MimeType.JSON);
  }
  
  // 2. 預設邏輯：讀取或建立帳號自選股
  const username = e.parameter.username;
  if (!username) {
    return ContentService.createTextOutput(JSON.stringify({ error: 'No username or action provided' }))
                         .setMimeType(ContentService.MimeType.JSON);
  }

  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
  const data = sheet.getDataRange().getValues();
  
  // 預設四個獨立區塊的標的與空倉
  const defaultWatchlist = {
    watchlist: ['AAPL', 'TSLA', 'NVDA'],
    watchlist_tw: ['2330.TW', '2317.TW'],
    crypto: ['BTCUSDT', 'ETHUSDT', 'SOLUSDT'],
    macro: ['^GSPC', '^IXIC', '^SOX'],
    forex: ['TWD=X', 'JPY=X', 'EURUSD=X'],
    portfolio: {},
    customNames: {}
  };

  // 尋找使用者
  for (let i = 1; i < data.length; i++) {
    if (data[i][0] === username) {
      let watchlistVal = data[i][1];
      let parsed;
      try {
        parsed = JSON.parse(watchlistVal || '{}');
      } catch(e) {
        parsed = {};
      }
      
      // 相容性檢查：若資料是舊版陣列，自動升級為新版物件格式
      if (Array.isArray(parsed)) {
        parsed = {
          watchlist: parsed,
          watchlist_tw: defaultWatchlist.watchlist_tw,
          crypto: defaultWatchlist.crypto,
          macro: defaultWatchlist.macro,
          forex: defaultWatchlist.forex,
          portfolio: {},
          customNames: {}
        };
        sheet.getRange(i + 1, 2).setValue(JSON.stringify(parsed));
      } else {
        parsed.watchlist = parsed.watchlist || defaultWatchlist.watchlist;
        parsed.watchlist_tw = parsed.watchlist_tw || defaultWatchlist.watchlist_tw;
        parsed.crypto = parsed.crypto || defaultWatchlist.crypto;
        parsed.macro = parsed.macro || defaultWatchlist.macro;
        parsed.forex = parsed.forex || defaultWatchlist.forex;
        parsed.portfolio = parsed.portfolio || {};
        parsed.customNames = parsed.customNames || {};
      }

      return ContentService.createTextOutput(JSON.stringify({ 
        status: 'success', 
        watchlist: parsed 
      })).setMimeType(ContentService.MimeType.JSON);
    }
  }
  
  // 如果是全新使用者，建立預設監控清單
  sheet.appendRow([username, JSON.stringify(defaultWatchlist)]);
  return ContentService.createTextOutput(JSON.stringify({ 
    status: 'created', 
    watchlist: defaultWatchlist 
  })).setMimeType(ContentService.MimeType.JSON);
}

// 處理前端發送的 POST 請求 (更新自選清單與虛擬持倉)
function doPost(e) {
  setup();
  const params = JSON.parse(e.postData.contents);
  const username = params.username;
  const watchlist = params.watchlist;

  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
  const data = sheet.getDataRange().getValues();

  for (let i = 1; i < data.length; i++) {
    if (data[i][0] === username) {
      sheet.getRange(i + 1, 2).setValue(JSON.stringify(watchlist));
      return ContentService.createTextOutput(JSON.stringify({ status: 'updated' }))
                           .setMimeType(ContentService.MimeType.JSON);
    }
  }
  
  return ContentService.createTextOutput(JSON.stringify({ error: 'User not found' }))
                       .setMimeType(ContentService.MimeType.JSON);
}
